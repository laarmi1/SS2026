Panduan Integrasi Google Sheets (100% Gratis)

Agar tombol "Kirim ke Google Sheets" secara otomatis membuat **Sheet/Tab Baru Sesuai Tanggal Hari Ini (contoh: 22-08-2026)**, ikuti langkah berikut:

  1. Buka Google Sheets Anda. Pilih menu Extensions (Ekstensi) > Apps Script.
  2. Ganti semua kode bawaan dengan **Copy & Paste kode script** terbaru di bawah ini:
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    
    // Gunakan tanggal dari pengiriman data atau tanggal sistem hari ini (Format: DD-MM-YYYY)
    var dateStr = data.dateStr || Utilities.formatDate(new Date(), ss.getSpreadsheetTimeZone(), "dd-MM-yyyy");
    
    // Cari tab sheet sesuai nama tanggal hari ini, jika belum ada buat tab baru!
    var sheet = ss.getSheetByName(dateStr);
    if (!sheet) {
      sheet = ss.insertSheet(dateStr);
    }
    
    // Tambah Header jika tab sheet masih kosong baru dibuat
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["No", "Nomor Resi", "Kurir", "Waktu Scan", "Batch", "Keterangan"]);
      sheet.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#e0f2fe");
      sheet.setFrozenRows(1);
    }
    
    var items = data.items;
    var startNo = sheet.getLastRow() === 1 ? 1 : sheet.getLastRow();
    for (var i = 0; i < items.length; i++) {
      var item = items[i];
      sheet.appendRow([
        startNo + i,
        item.resi,
        item.courier,
        item.timestamp,
        data.batchName || "Batch",
        item.note || "-"
      ]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({result: "success", count: items.length, sheet: dateStr}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
  1. Klik Deploy (Terapkan) > New Deployment (Penerapan baru) (atau Manage Deployments lalu edit/versi baru).
  2. Pilih jenis Web App (Aplikasi Web).
    • Execute as: Me (Saya)
    • Who has access: Anyone (Siapa saja)
  3. Klik Deploy, izinkan akses Google, lalu pastekan Web App URL ke menu Pengaturan web ini. Selesai!
